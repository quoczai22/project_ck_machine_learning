class ZeroToMissingTransformer(BaseEstimator, TransformerMixin):
    """
    Chuyển 0 -> NaN trên 5 cột sinh lý.
    Giữ nguyên 0 ở Pregnancies.

    Dữ liệu từ Việc 5 đã có NaN sẵn.
    Transformer vẫn được giữ để xử lý dữ liệu mới.
    """

    def __init__(self, cols_to_convert):
        self.cols_to_convert = cols_to_convert

    def fit(self, X, y=None):
        return self

    def transform(self, X):

        X_copy = X.copy()

        for col in self.cols_to_convert:

            if col in X_copy.columns:
                X_copy[col] = X_copy[col].replace(0, np.nan)

        return X_copy


zero_cols = [
    "Glucose",
    "BloodPressure",
    "SkinThickness",
    "Insulin",
    "BMI"
]


# Random Forest chỉ được sử dụng làm mô hình minh họa
# trong quá trình kiểm tra Pipeline và Data Leakage.
