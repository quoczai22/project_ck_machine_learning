print("\n>>> Chon file pima_indians_zero_as_missing_NaN.csv")
uploaded = files.upload()

if not uploaded:
    raise RuntimeError("Chua chon file CSV!")

filename = next(iter(uploaded.keys()))
input_file_bytes = uploaded[filename]

# Ghi nhan thoi diem bat dau xu ly sau khi chon file.
run_started_at = datetime.now().astimezone().isoformat(timespec="seconds")
start_time = time.perf_counter()

# SHA-256 cua file CSV goc, tinh tren bytes duoc upload.
dataset_sha256 = hashlib.sha256(input_file_bytes).hexdigest()

df = pd.read_csv(io.BytesIO(input_file_bytes))
print(f">>> Da doc: {df.shape[0]} dong, {df.shape[1]} cot")
print(f">>> Dataset SHA-256: {dataset_sha256}")


# ======================================================================
# CHUAN HOA TEN COT MUC TIEU THANH OUTCOME
# ======================================================================

if "Class" in df.columns and "Outcome" not in df.columns:
    df = df.rename(columns={"Class": "Outcome"})
    print(">>> Da doi ten cot Class thanh Outcome.")
elif "Class" in df.columns and "Outcome" in df.columns:
    raise ValueError(
        "Dataset dang co ca Class va Outcome. Vui long kiem tra lai du lieu."
    )
elif "Outcome" not in df.columns:
    raise ValueError("Khong tim thay cot muc tieu Class hoac Outcome.")

TARGET_COLUMN = "Outcome"
print(f">>> Target column su dung: {TARGET_COLUMN}")

if df[TARGET_COLUMN].isna().any():
    raise ValueError("Cot Outcome co gia tri thieu (NaN).")

if not set(df[TARGET_COLUMN].unique()).issubset({0, 1}):
    raise ValueError("Cot Outcome chi duoc chua gia tri 0 va 1.")


# ======================================================================
# KIEM TRA DU LIEU DA XU LY 0 -> NaN
# ======================================================================

zero_cols_check = [
    "Glucose",
    "BloodPressure",
    "SkinThickness",
    "Insulin",
    "BMI",
]

missing_columns = [col for col in zero_cols_check if col not in df.columns]
if missing_columns:
    raise ValueError(f"Thieu cac cot sinh ly: {missing_columns}")

nan_counts = df[zero_cols_check].isna().sum()
remaining_zeros = int((df[zero_cols_check] == 0).sum().sum())

print("\n[KIEM TRA DU LIEU DA XU LY]")
print("-" * 60)
print(nan_counts.to_string())
print(f"Tong so NaN: {int(nan_counts.sum())}")
print(f"So gia tri 0 con lai: {remaining_zeros}")

if remaining_zeros != 0:
    raise AssertionError(
        "File chua duoc xu ly 0 -> NaN. Vui long dung file tu Viec 5."
    )
print("PASS: Du lieu da xu ly 0 -> NaN.")


# Luu dataset da chuan hoa ten cot Outcome.
normalized_file = "pima_indians_zero_as_missing_NaN_Outcome.csv"
df.to_csv(normalized_file, index=False, encoding="utf-8-sig")
print(f">>> Da luu dataset chuan hoa: {normalized_file}")
