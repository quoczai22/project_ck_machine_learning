# ---- 1.2. Chia Train/Test ----
X_train, X_test, y_train, y_test = train_test_split(

# ---- 1.3. Custom Transformer: Zero -> Missing (để handle dữ liệu mới) ----
class ZeroToMissingTransformer(BaseEstimator, TransformerMixin):
    """
    Chuyển 0 -> NaN trên 5 cột sinh lý. Giữ nguyên 0 ở Pregnancies.

    Lưu ý: Dữ liệu từ Việc 5 đã có NaN sẵn → bước này không làm gì.
    Nhưng vẫn cần giữ để xử lý DỮ LIỆU MỚI (bệnh nhân mới có thể nhập số 0).
    """
    def __init__(self, cols_to_convert):
        self.cols_to_convert = cols_to_convert

    def fit(self, X, y=None):
        return self

    def transform(self, X):
        X_copy = X.copy()
        for col in self.cols_to_convert:
            if col in X_copy.columns:
                X_copy[col] = X_copy[col].replace(0, np.nan)
        return X_copy

# ---- 1.4. Xây dựng Pipeline 4 bước ----
zero_cols = ['Glucose', 'BloodPressure', 'SkinThickness', 'Insulin', 'BMI']

preprocessing_pipeline = Pipeline(steps=[
    ('zero_to_missing', ZeroToMissingTransformer(cols_to_convert=zero_cols)),
    ('imputer',         SimpleImputer(strategy='median')),
    ('scaler',          StandardScaler()),
    ('classifier',      RandomForestClassifier(random_state=42))
])
